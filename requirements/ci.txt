# Python dependencies for continuous integration test runs.
#
# Versions are pinned exactly rather than floored like requirements/harness.txt.
# CI judges a pull request by comparing the tests that are red now against
# tests/ci_baseline.json. A dependency that resolved to a different version than
# the one the baseline was recorded against would surface as a test changing
# state, which is indistinguishable from a real regression. The resolution has
# to be fixed, not merely compatible.
#
# Bump deliberately, in its own pull request, so the effect is attributable, and
# regenerate the baseline in that same pull request if the bump moves any test.

# Test runner.
pytest==9.1.1

# Caps a single hung test rather than letting it consume the whole job
# budget. At least one test drives a subprocess that waits on a provider
# that is not there; without a per-test ceiling that one test can take the
# shard's entire timeout and the shard then reports nothing.
pytest-timeout==2.4.0

# Imported by the suite. Without flask, four test modules fail at collection
# time, which makes pytest exit 2 and takes the whole run down rather than
