#!/usr/bin/env bash
set -e

config_value() {
    key="$1"
    file="$2"
    [ -f "$file" ] || return 1
    while IFS= read -r line; do
        case "$line" in
            "$key="*) printf '%s\n' "${line#*=}"; return 0 ;;
            "export $key="*) printf '%s\n' "${line#*=}"; return 0 ;;
        esac
    done < "$file"
    return 1
}

receipt_value() {
    file="$1"
    key="$2"
    [ -f "$file" ] || return 1
    command -v python3 >/dev/null 2>&1 || return 1
    python3 - "$file" "$key" <<'PY'
import json
import sys

with open(sys.argv[1], encoding="utf-8") as f:
    value = json.load(f).get(sys.argv[2], "")
if value:
    print(value)
else:
    raise SystemExit(1)
PY
}

resolve_script_dir() {
    script="$0"
    case "$script" in
        */*) ;;
        *)
            found="$(command -v -- "$script" 2>/dev/null || true)"
            [ -n "$found" ] && script="$found"
            ;;
    esac
    while [ -L "$script" ]; do
        script_dir="$(CDPATH= cd -- "$(dirname -- "$script")" && pwd -P)"
        link="$(readlink "$script")" || break
        case "$link" in
            /*) script="$link" ;;
            *) script="$script_dir/$link" ;;
        esac
    done
    CDPATH= cd -- "$(dirname -- "$script")" && pwd -P
}

SCRIPT_DIR="$(resolve_script_dir)"
SCRIPT_SOLAR_HOME=""
if [ "$(basename -- "$SCRIPT_DIR")" = "bin" ]; then
    candidate_home="$(CDPATH= cd -- "$SCRIPT_DIR/.." && pwd -P)"
    if [ -f "$candidate_home/install-receipt.json" ] || [ -f "$candidate_home/config.env" ]; then
        SCRIPT_SOLAR_HOME="$candidate_home"
    fi
fi

SOLAR_HOME="${SOLAR_HOME:-${SCRIPT_SOLAR_HOME:-$HOME/.solar}}"
RECEIPT_PATH="$SOLAR_HOME/install-receipt.json"
if [ -z "${CLAUDE_DIR:-}" ]; then
    CLAUDE_DIR="$(config_value CLAUDE_DIR "$SOLAR_HOME/config.env" 2>/dev/null || true)"
fi
if [ -z "${CLAUDE_DIR:-}" ]; then
    CLAUDE_DIR="$(receipt_value "$RECEIPT_PATH" claude_dir 2>/dev/null || true)"
fi
CLAUDE_DIR="${CLAUDE_DIR:-$HOME/.claude}"

    cat <<'EOF'
solar - OpenSolar lifecycle command

Usage:
  solar version [--json]
  solar status [--json]

  solar update [extra install flags]
  solar repair [extra install flags]

  solar restore <archive>
  solar ui [--once] [--watch N] [--no-color]
  solar harness [verb] [args...]
  solar uninstall [--yes] [--keep-data] [--dry-run]
  solar components list
EOF
}

receipt_field() {
    [ -f "$RECEIPT_PATH" ] || return 1
    RECEIPT_PATH="$RECEIPT_PATH" python3 - "$1" <<'PY'
import json
import os
import sys

with open(os.environ["RECEIPT_PATH"], encoding="utf-8") as f:
    receipt = json.load(f)
value = receipt.get(sys.argv[1], "")
if isinstance(value, list):
    print(",".join(value))
else:
    print(value)
PY
}

# Data paths under SOLAR_HOME that backup/restore and --keep-data treat as
# user data (config + secrets + receipt + database).
SOLAR_DATA_PATHS="install-receipt.json config.env .env db"

do_version() {
    json=false
    [ "${1:-}" = "--json" ] && json=true
    if [ ! -f "$RECEIPT_PATH" ]; then
        echo "no install receipt at $RECEIPT_PATH; run install.sh first" >&2
        exit 1
    fi
    ver="$(receipt_field version)"; [ -n "$ver" ] || ver="unknown"
    sha="$(receipt_field git_sha)"
    chan="$(receipt_field channel)"
    repo="$(receipt_field repo)"
    if [ "$json" = "true" ]; then
        RV="$ver" RS="$sha" RC="$chan" RR="$repo" python3 - <<'PY'
import json
import os

print(json.dumps({
    "version": os.environ.get("RV", ""),
    "git_sha": os.environ.get("RS", ""),
    "channel": os.environ.get("RC", ""),
    "repo": os.environ.get("RR", ""),
}, indent=2, sort_keys=True))
PY
    else
        printf 'OpenSolar %s' "$ver"
        if [ -n "$sha" ] && [ "$sha" != "unknown" ]; then
            printf ' (git %s)' "$sha"
        fi
        printf '\n'
        # A channel-less (pre-batch-2) receipt must not make the last command
        # false; that would propagate exit 1 under set -e despite valid output.
        if [ -n "$chan" ]; then
            printf 'channel: %s\n' "$chan"
        fi
    fi
}

do_backup() {
    out=""
    while [ "$#" -gt 0 ]; do
        case "$1" in
            --out) out="$2"; shift 2 ;;
            *) echo "unknown backup option: $1" >&2; exit 2 ;;
        esac
    done
    [ -d "$SOLAR_HOME" ] || { echo "no Solar home at $SOLAR_HOME" >&2; exit 1; }
    set --
    for p in $SOLAR_DATA_PATHS; do
        [ -e "$SOLAR_HOME/$p" ] && set -- "$@" "$p"
    done
    [ "$#" -gt 0 ] || { echo "nothing to back up in $SOLAR_HOME" >&2; exit 1; }
    if [ -z "$out" ]; then
        mkdir -p "$SOLAR_HOME/backups"
        out="$SOLAR_HOME/backups/solar-backup-$(date -u +%Y%m%d%H%M%S).tar.gz"
    fi
    tar czf "$out" -C "$SOLAR_HOME" "$@"
    echo "backup written: $out"
}

do_status() {
    json=false
    while [ "$#" -gt 0 ]; do
        case "$1" in
            --json) json=true; shift ;;
            --help|-h) echo "usage: solar status [--json]"; return 0 ;;
            *) echo "unknown status option: $1" >&2; exit 2 ;;
        esac
    done
    SOLAR_HOME="$SOLAR_HOME" CLAUDE_DIR="$CLAUDE_DIR" RECEIPT_PATH="$RECEIPT_PATH" \
    STATUS_JSON="$json" python3 - <<'PY'
import http.client
import json
import os
import socket
import subprocess
import sys
from pathlib import Path


def load_json(path):
    try:
        with open(path, encoding="utf-8") as f:
            return json.load(f)
    except Exception:
        return {}


def run_cmd(argv, timeout=5):
    try:
        proc = subprocess.run(
            argv,
            check=False,
            text=True,
            capture_output=True,
            timeout=timeout,
        )
        return proc.returncode, proc.stdout, proc.stderr
    except FileNotFoundError as exc:
        return 127, "", str(exc)
    except subprocess.TimeoutExpired as exc:
        out = exc.stdout if isinstance(exc.stdout, str) else ""
        err = exc.stderr if isinstance(exc.stderr, str) else ""
        return 124, out, err or f"timed out after {timeout}s"


def path_state(path, want_dir=False):
    ok = path.is_dir() if want_dir else path.exists()
    return "ok" if ok else "missing"


def live_pid(raw):
    text = str(raw or "").strip()
    if not text.isdigit():
        return False
    try:
        os.kill(int(text), 0)
        return True
    except OSError:
        return False


def tmux_session(name):
    rc, _, err = run_cmd(["tmux", "has-session", "-t", name], timeout=2)
    if rc == 0:
        return {"state": "present", "detail": name}
    if rc == 127:
        return {"state": "unavailable", "detail": "tmux not on PATH"}
    detail = (err or "").strip()
    return {"state": "absent", "detail": detail or name}


def recent_runtime_files(harness_dir):
    roots = [
        harness_dir / "run" / "operator-inbox",
        harness_dir / "run" / "operator-results",
        harness_dir / "run" / "operator-status",
        harness_dir / "run" / "queue",
        harness_dir / "sprints",
    ]
    found = []
    for root in roots:
        if not root.exists():
            continue
        for pattern in ("*.json", "*.jsonl", "*.status.json"):
            found.extend(p for p in root.rglob(pattern) if p.is_file())
    found.sort(key=lambda p: p.stat().st_mtime if p.exists() else 0, reverse=True)
    out = []
    for path in found[:5]:
        try:
            out.append(str(path.relative_to(harness_dir)))
        except ValueError:
            out.append(str(path))
    return out


def daemon_health(socket_path):
    if not socket_path.exists():
        return {"state": "not-running", "detail": f"socket missing: {socket_path}"}
    try:
        conn = http.client.HTTPConnection("localhost", timeout=1)
        sock = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
        sock.settimeout(1)
        sock.connect(str(socket_path))
        conn.sock = sock
        conn.request("GET", "/health")
        resp = conn.getresponse()
        body = resp.read().decode("utf-8", "replace")
        conn.close()
        if resp.status == 200:
            try:
                payload = json.loads(body)
            except Exception:
                payload = {"raw": body.strip()}
            return {"state": "running", "detail": payload}
        return {"state": "error", "detail": f"HTTP {resp.status}: {body.strip()}"}
    except Exception as exc:
        return {"state": "error", "detail": str(exc)}
